// QA20261020: editable closed-bar SMA; no network, files, orders or hidden state.
function decimalMean(values) {
const parts = values.map(value => {
const match = /^(-?)([0-9]+)(?:\.([0-9]+))?$/.exec(String(value));
if (!match) throw new Error('Invalid Decimal close; never replace missing input with zero');
return {negative: match[1] === '-', whole: match[2], fraction: match[3] || ''};
});
const scale = Math.max(8, ...parts.map(part => part.fraction.length));
const total = parts.reduce((sum, part) => {
const amount = BigInt(part.whole + part.fraction.padEnd(scale, '0'));
return sum + (part.negative ? -amount : amount);
}, 0n);
const denominator = BigInt(values.length);
const negative = total < 0n;
const absolute = negative ? -total : total;
let quotient = absolute / denominator;
const remainder = absolute % denominator;
if (remainder * 2n > denominator || (remainder * 2n === denominator && quotient % 2n === 1n)) quotient += 1n;
const digits = quotient.toString().padStart(scale + 1, '0');
const fraction = digits.slice(-scale).replace(/0+$/, '');
return (negative && quotient !== 0n ? '-' : '') + digits.slice(0, -scale) + (fraction ? '.' + fraction : '');
}
export function compute(input) {
const period = input.parameters.period === undefined ? 20 : input.parameters.period;
if (!Number.isInteger(period) || period < 2 || period > 100) throw new Error('period must be an integer from 2 to 100');
const rows = input.inputs.price.rows.filter(bar => bar.isClosed === true);
const points = [];
for (let index = period - 1; index < rows.length; index += 1) {
const bar = rows[index];
const value = decimalMean(rows.slice(index - period + 1, index + 1).map(row => row.close));
points.push({key: bar.id, time: bar.openTime, value: {status: 'value', value}});
}
return {status: points.length ? 'ready' : 'not_ready', series: [{seriesId: 'sma', points}], diagnostics: []};
}
